% Edge tabs: one tab per part (or topic) on the right edge of every body page: light grey, except the current
% one, which has the same size and is in its part's colour (the same colour as on the cover); each tab is a link to its
% part. build.py loads this file when the build has --edge-tabs (theory notes, exam notes and formula sheets
% by default) and fills \HouseTabList with \housetabitem{n}{label} items. The current part is the last one
% whose label is on this page or before it (page references, so it is right after the second LaTeX run).
% The print copy has no tabs (printers cannot print to the edge of the sheet).

% Chip sizes (cm): every chip is the same width (the colour alone marks the current part); \housetabgap is the
% space between two chips
\def\housetabwide{0.4}\def\housetabnarrow{0.4}\def\housetabgap{0.06}
% A chip: the fill, then a link annotation exactly the chip's size (XeTeX sizes a \hyperref by the glyphs inside
% it, which would leave only the number clickable)
\newcommand{\housetabchip}[3]{% {fill colour}{width cm}{label}
	\fill[#1] ($(current page.north east)+(-#2,-\housetabtop)$) rectangle ++(#2,-\housetabh);
	\node[inner sep=0pt, outer sep=0pt] at ($(current page.north east)+(-#2,-\housetabtop-\housetabh)$)
		{\special{pdf:ann width #2cm height \housetabh cm depth 0pt
			<< /Type /Annot /Subtype /Link /Border [0 0 0] /A << /S /GoTo /D (#3) >> >>}};}
\newcommand{\housetabitem}[2]{%
	\setpartcolor{#1}%
	\pgfmathsetmacro{\housetabtop}{2.6 + (#1-1)*\housetabstep}%
	\pgfmathsetmacro{\housetabh}{\housetabstep-\housetabgap}%
	\ifnum#1=\housecurrent\relax
		\housetabchip{\partcolor}{\housetabwide}{#2}%
		\node[rotate=-90, text=white, font=\bfseries\scriptsize, anchor=center]
			at ($(current page.north east)+(-\housetabwide/2,-\housetabtop-\housetabh/2)$) {#1};
	\else
		\housetabchip{gray!18}{\housetabnarrow}{#2}%
		\node[rotate=-90, text=gray!55!black, font=\scriptsize, anchor=center]
			at ($(current page.north east)+(-\housetabnarrow/2,-\housetabtop-\housetabh/2)$) {#1};
	\fi}
% first pass over the list: find the current part
\newcommand{\housetabscan}[2]{\ifnum\getpagerefnumber{#2}>\value{page}\else\def\housecurrent{#1}\fi}
\AddToHook{shipout/background}{\ifHouseTabs
	\def\housecurrent{0}%
	\begingroup\let\housetabitem\housetabscan\HouseTabList\expandafter\endgroup\expandafter\def\expandafter\housecurrent\expandafter{\housecurrent}%
	\put(0,0){\begin{tikzpicture}[remember picture, overlay, x=1cm, y=1cm]
		\pgfmathsetmacro{\housetabstep}{min(1.6, (\paperheight/1cm - 5.2)/\HouseTabCount)}%
		\HouseTabList
	\end{tikzpicture}}\fi}
