% Share copy: build.py --share (and bundle.py --share) loads this after the house and type preambles.
% A copy for sharing that is awkward to print and resell: build.py locks printing in the PDF afterwards, and
% this file adds what still works when a viewer ignores that lock.
%   - a small line in the footer of every page but the covers, always visible (it survives screenshots and
%     every viewer)
%   - the red and blue headings, a step lighter, on a layer that shows on screen and is never printed: a
%     print has blank space where they were
%   - a large diagonal stamp on a layer that is hidden on screen and always printed
%   Both layers start visible and say how they behave in /Usage (view, print). Acrobat, Foxit, PDF-XChange,
%   Chrome and Firefox's viewer follow that; Firefox can hide a layer in print but never show one that starts
%   hidden, which is why the stamp starts visible. A viewer that ignores layers altogether shows the stamp
%   (light grey) on screen and prints the headings.
% The layers are written with xdvipdfmx specials, not ocgx2: ocgx2 carries a layer that is open at a page break
% over to the next page, and the stamp (drawn while the page is shipped out) confused that, so a whole page
% landed on the headings layer. A heading never breaks across pages, so each mark opens and closes on one page.
% \HouseShareText (the text of the footer line and the stamp) is defined by the build before this file is read.
% The normal and print builds never use it.

\def\HouseShareCopy{}

% Headings: a step lighter than the house red!70!black and blue!75!black (they show as about 60 % grey)
\definecolor{headred}{HTML}{BF4040}
\definecolor{headblue}{HTML}{4560D0}

% ---------- The two layers (optional content groups) ----------
\AddToHook{shipout/firstpage}{%
	\special{pdf:obj @househeadings << /Type /OCG /Name (Headings, screen only)
		/Usage << /View << /ViewState /ON >> /Print << /PrintState /OFF >> >> >>}%
	\special{pdf:obj @housestamp << /Type /OCG /Name (Print stamp)
		/Usage << /View << /ViewState /OFF >> /Print << /PrintState /ON >> >> >>}%
	\special{pdf:put @catalog << /OCProperties << /OCGs [@househeadings @housestamp]
		/D << /Order [@househeadings @housestamp] /BaseState /ON
		/AS [<< /Event /View /Category [/View] /OCGs [@househeadings @housestamp] >>
		     << /Event /Print /Category [/Print] /OCGs [@househeadings @housestamp] >>
		     << /Event /Export /Category [/Export] /OCGs [@househeadings @housestamp] >>] >> >> >>}}
% every page names both layers in its resources, so its marks can point at them
\AddToHook{shipout/background}{\special{pdf:put @resources << /Properties << /HouseHead @househeadings /HouseStamp @housestamp >> >>}}
% a heading's title can arrive between paragraphs: start its line first (as its first letter would), so the
% mark stays on the heading's line and a page break cannot fall between them
\newcommand{\HouseLayer}[2]{\ifvmode\leavevmode\fi\special{pdf:literal direct /OC /#1 BDC}#2\special{pdf:literal direct EMC}}

% each heading's number and title: on the screen-only layer
\renewcommand{\headingwrap}[1]{\HouseLayer{HouseHead}{#1}}

% ---------- Footer line ----------
% right of the page number on every page with a footer (covers have none, so it stays off their artwork);
% plain is the house footer again, with the line
\newcommand{\HouseShareLine}{{\normalfont\scriptsize\color{black!55}\HouseShareText}}
\fancyfoot[R]{\HouseShareLine}
\fancypagestyle{plain}{\fancyhf{}\renewcommand{\headrulewidth}{0pt}\fancyfoot[C]{\thepage}\fancyfoot[R]{\HouseShareLine}}

% ---------- Print stamp ----------
\newcommand{\HouseShareStamp}{%
	\put(0.5\paperwidth,-0.5\paperheight){\HouseLayer{HouseStamp}{%
		\rotatebox{\ifdim\paperwidth>\paperheight 35\else 55\fi}{\makebox[0pt]{\normalfont\fontsize{40}{40}\selectfont
			\bfseries\color{black!30}\raisebox{-0.35em}{\HouseShareText}}}}}}
\AddToHook{shipout/foreground}{\HouseShareStamp}
