% Share copy: build.py --share (and bundle.py --share) loads this after the house and type preambles.
% A copy for sharing that is awkward to print and resell: build.py locks printing in the PDF afterwards, and
% this file adds what still shows when a viewer ignores that lock.
%   - a small line in the footer of every page but the covers, always visible (it survives screenshots and
%     every viewer)
%   - a large diagonal stamp on a layer that is hidden on screen and always printed (Acrobat, Foxit,
%     PDF-XChange; a viewer that ignores layers does not print it, so the footer line carries the name there)
%   - the red and blue headings in a slightly lighter pair
% \HouseShareText (the text of both) is defined by the build before this file is read.
% The normal and print builds never use it.

\def\HouseShareCopy{}

% Headings: a step lighter than the house red!70!black and blue!75!black (they print as about 60 % grey)
\definecolor{headred}{HTML}{BF4040}
\definecolor{headblue}{HTML}{4560D0}

\usepackage{ocgx2}
% the footer line: right of the page number on every page with a footer (covers have none, so it stays off
% their artwork); plain is the house footer again, with the line
\newcommand{\HouseShareLine}{{\normalfont\scriptsize\color{black!55}\HouseShareText}}
\fancyfoot[R]{\HouseShareLine}
\fancypagestyle{plain}{\fancyhf{}\renewcommand{\headrulewidth}{0pt}\fancyfoot[C]{\thepage}\fancyfoot[R]{\HouseShareLine}}
\newcommand{\HouseShareStamp}{%
	\put(0.5\paperwidth,-0.5\paperheight){%
		\begin{ocg}[viewocg=never, printocg=always]{Print stamp}{housesharestamp}{0}%
		\rotatebox{\ifdim\paperwidth>\paperheight 35\else 55\fi}{\makebox[0pt]{\normalfont\fontsize{40}{40}\selectfont
			\bfseries\color{black!30}\raisebox{-0.35em}{\HouseShareText}}}%
		\end{ocg}}}
\AddToHook{shipout/foreground}{\HouseShareStamp}
