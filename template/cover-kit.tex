		\draw[CoverSig, opacity=\op, line width=1pt, domain=\s:\s+0.52, samples=12, smooth]
			plot ({\x}, {9.8+\k*0.75 + 0.35*sin(deg(\x*(0.9+\k*0.12)+\k*40))*(0.5+0.5*sin(deg(\x*0.22)))});}\end{scope}}

% ---------- print copy: page order for double-sided printing ----------
% The back of the cover is page 2 (a blank page when the type has none), so the content starts on a
% right-hand page. \HousePadBefore{n}: blank pages until the next page is the last of a group of n, for the
% back cover (n = 2: the back of the last sheet; n = 4: the back of a folded booklet). \HousePadTo{n}: blank
% pages until the page count is a multiple of n (no back cover).
\newcommand{\HouseBlankPage}{\null\thispagestyle{empty}\clearpage}
\newcommand{\HousePadBefore}[1]{\clearpage
	\pgfmathtruncatemacro{\housepadleft}{mod(\ReadonlyShipoutCounter + 1, #1)}%
	\ifnum\housepadleft>0 \HouseBlankPage\HousePadBefore{#1}\fi}
\newcommand{\HousePadTo}[1]{\clearpage
	\pgfmathtruncatemacro{\housepadleft}{mod(\ReadonlyShipoutCounter, #1)}%
	\ifnum\housepadleft>0 \HouseBlankPage\HousePadTo{#1}\fi}
