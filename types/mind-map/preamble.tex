% Links the size of what they cover. XeTeX makes a \hyperref exactly as big as the glyphs inside it, so a button
% would only be clickable on its letters; these write the link annotation themselves.
\newsavebox\mmlb
% \mmlinkbox{target}{content}: the whole box of the content is the link (no link when the target is empty)
\newcommand\mmlinkbox[2]{\leavevmode\ifstrempty{#1}{#2}{\sbox\mmlb{#2}\special{pdf:ann width \the\wd\mmlb\space height \the\ht\mmlb\space
	depth \the\dp\mmlb\space << /Type /Annot /Subtype /Link /Border [0 0 0] /A << /S /GoTo /D (#1) >> >>}\usebox\mmlb}}
% \mmannot{width}{height}{target}, inside a tikzpicture: a link with its lower-left corner at (0,0)
\newcommand\mmannot[3]{\node[inner sep=0pt, outer sep=0pt] at (0,0) {\special{pdf:ann width #1 height #2 depth 0pt
	<< /Type /Annot /Subtype /Link /Border [0 0 0] /A << /S /GoTo /D (#3) >> >>}};}
% \mmnodelink{node}{target}: the whole node is a link
\newcommand\mmnodelink[2]{%
	\path (#1.north east); \pgfgetlastxy{\mmxa}{\mmya}%
	\path (#1.south west); \pgfgetlastxy{\mmxb}{\mmyb}%
	\pgfmathsetlengthmacro\mmw{\mmxa-\mmxb}\pgfmathsetlengthmacro\mmh{\mmya-\mmyb}%
	\begin{scope}[shift={(\mmxb,\mmyb)}]\mmannot{\mmw}{\mmh}{#2}\end{scope}}

% The button strip on a card's right edge: the definition button (top, or all of it), and the sub-map button
% (bottom half) when the card has a sub-map. \mmifont, \mmisep and \mmmfont set the sizes per paper.
\def\mmifont{\scriptsize}\def\mmisep{0.5pt}\def\mmmfont{\scriptsize}
% \mmstrip{card node}{part}{strip width cm}{definition target}{sub-map target or empty}{count}
\newcommand\mmstrip[6]{%
	\path (#1.north east); \pgfgetlastxy{\mmxa}{\mmya}%
	\path (#1.south east); \pgfgetlastxy{\mmxb}{\mmyb}%
	\pgfmathsetlengthmacro\mmsw{#3cm}%
	\pgfmathsetlengthmacro\mmh{\mmya-\mmyb}%
	\ifstrempty{#5}{\pgfmathsetlengthmacro\mmhi{\mmh}}{\pgfmathsetlengthmacro\mmhi{\mmh/2}}%
	\begin{scope}
		\clip[rounded corners=4pt] (#1.north west) rectangle (#1.south east);
		\fill[CoverPart#2!13] (\mmxa-\mmsw,\mmya) rectangle (\mmxa,\mmya-\mmhi);
		\ifstrempty{#5}{}{\fill[CoverPart#2] (\mmxa-\mmsw,\mmyb) rectangle (\mmxa,\mmya-\mmhi);}
	\end{scope}
	\draw[CoverPart#2, line width=0.8pt, rounded corners=4pt] (#1.north west) rectangle (#1.south east);
	\draw[CoverPart#2!45, line width=0.5pt] (\mmxa-\mmsw,\mmya) -- (\mmxa-\mmsw,\mmyb);
	\node[circle, draw=CoverPart#2, fill=white, line width=0.6pt, inner sep=\mmisep, font=\mmifont\bfseries\itshape,
		text=CoverPart#2!80!black] at (\mmxa-\mmsw/2,\mmya-\mmhi/2) {i};
	\begin{scope}[shift={(\mmxa-\mmsw,\mmya-\mmhi)}]\mmannot{\mmsw}{\mmhi}{#4}\end{scope}
	\ifstrempty{#5}{}{%
		\node[font=\mmmfont\bfseries, text=white, align=center, inner sep=0pt] at (\mmxa-\mmsw/2,\mmyb+\mmh/4) {$\blacktriangleright$\!#6};
		\begin{scope}[shift={(\mmxa-\mmsw,\mmyb)}]\mmannot{\mmsw}{\mmhi}{#5}\end{scope}}}
