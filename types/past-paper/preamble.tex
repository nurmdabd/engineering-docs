% ---------- One solution: questionblock[TOC title]{full question text} ----------
% The full question is printed highlighted; the TOC shows the short title. Sections inside are numbered
% 1, 2, ... per solution. Every format change below is local to the environment, so headings outside
% it keep the house style.
\ifcsname c@shadowq\endcsname\else\newcounter{shadowq}\fi
\NewDocumentEnvironment{questionblock}{o m}
{
	\refstepcounter{shadowq}
	\phantomsection
	\titleformat{name=\section,numberless}{\normalfont\normalsize\bfseries}{}{0pt}{}
	\section*{\normalsize\grayhl{#2}}
	\IfNoValueTF{#1}
	{\addcontentsline{toc}{section}{#2}}
	{\addcontentsline{toc}{section}{#1}}

	\setcounter{subsection}{0}
	\renewcommand{\theHsubsection}{shadow.\theshadowq.\arabic{subsection}}
	\renewcommand{\thesubsection}{\arabic{subsection}}
	\titleformat{\subsection}
	{\Large\bfseries\color{headred}}
	{\thesubsection.}
	{0.3em}
	{}
	\titleformat{\subsubsection}
	{\large\bfseries\color{headblue}}
	{\thesubsubsection}
	{0.3em}
	{}
}
{}

% ---------- Misprint marker: [should be $\sin\theta$]{.sic} -> a small red [sic: ...] after the printed text ----------
% The printed text stays exactly as it is; the marker only flags it, and the solution's warning box explains it.
\newcommand{\sic}[1]{\unskip\,\mbox{\normalfont\scriptsize\color{red!75!black}[\textit{sic}\ifblank{#1}{}{: #1}]}}   % one unit: never split or stretched
\soulregister{\sic}{1}   % the question titles are highlighted with soul, which needs to know the argument

% ---------- The question paper (build.py --paper writes paper-page.tex from the mainquestion boxes) ----------
% Laid out like the printed paper: centred header, marks/credits/time row, the instruction, part headings,
% then each question with its marks in the right margin. Sub-part labels link to their solutions.
\newcommand{\currentpaperq}{paper}
\newcommand{\papersetq}[1]{\gdef\currentpaperq{paper-q#1}}   % each mainquestion box: PAPER jumps to its question
\NewDocumentEnvironment{questionpaper}{}
{
	\clearpage
	\gdef\HousePaperPage{}
	\phantomsection\hypertarget{paper}{}\pdfbookmark[0]{Question Paper}{paper}
	\begingroup\setstretch{1.05}\setlength{\parskip}{0pt}\setlength{\parindent}{0pt}
	\let\textbf\oldtextbf   % plain black bold, as on the printed paper
}
{
	\endgroup
	\clearpage
}
